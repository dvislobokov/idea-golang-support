"fmt"

type   point struct{
x,y int
    label string // the label
}

func (p point)   String() string{
return fmt.Sprintf("%s(%d,%d)",p.label,p.x,p.y)
}

func joinAll(items []string)string {
  var b strings.Builder
     for i,it:=range items{
  if i>0 {b.WriteString(", ")}
     b.WriteString(it)
  }
	return b.String()
}
