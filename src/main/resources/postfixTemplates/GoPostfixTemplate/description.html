Go postfix template: type the key after an expression and press Tab, or pick it in the completion list after . (the keys come after the fields and methods). A key is offered only where it fits the type of the expression; the tree shows what each one writes.
.if .elseif expr {} / if !expr {} (bool)
.nil .nn .notnilif expr == nil {} / if expr != nil {} (pointers, slices, maps, channels, functions, interfaces)
.err .errv .errncheck the error of a call or of an error value
.return .rrreturn expr (other results get zero values) / return expr, nil
.varname := expr, the name from the expression: area := c.Area()
.for .forr .rangefor _, v := range expr {} / for i, name := range names {} / for range expr {}
.fori .forrevfor i := 0; i < len(expr); i++ {} / from the last index down
.len .caplen(expr) / cap(expr)
.print .printf .printlnfmt.Println(expr) / fmt.Printf("%v\n", expr) / println(expr)
.panic .go .deferpanic(expr) / go call / defer call
.append .aappend .appendAssignappend(xs, …) / xs = append(xs, …) (slices)
.copy .removecopy(xs, …) / xs = append(xs[:i], xs[i+1:]...) (slices)
.close .deleteclose(ch) (channels) / delete(m, key) (maps)
.complex .real .imagcomplex(f, …) (floats) / real(z), imag(z) (complex numbers)
.not .!!expr (bool)
.& .p .pointer&expr
.* .d .dereference*expr (pointers)
.par .switch(expr) / switch expr {}
.wrap .as .isfmt.Errorf("…: %w", err) / errors.As(err, &target) / errors.Is(err, target) (errors)
.parseInt .parseFloatn, err := strconv.ParseInt(s, 10, 64) / strconv.ParseFloat(s, 64) with the check of the error (strings)
.sortsort.Strings / sort.Ints / sort.Float64s, sort.Sort for a sort.Interface, slices.Sort, sort.Slice
